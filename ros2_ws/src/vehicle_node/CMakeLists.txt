cmake_minimum_required(VERSION 3.8)
project(vehicle_node)
# ROS 2 Humble 기본 C++ 표준을 사용합니다.
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
# 제어 노드 하나를 빌드하고 ros2 run에서 찾을 위치에 설치합니다.
add_executable(vehicle_node src/vehicle_node.cpp)
ament_target_dependencies(vehicle_node rclcpp geometry_msgs nav_msgs tf2_ros)
install(TARGETS vehicle_node DESTINATION lib/${PROJECT_NAME})
# 사람이 Gazebo에서 움직임을 보는 별도 C++ 시연 프로그램입니다.
add_executable(driving_demo src/driving_demo.cpp)
ament_target_dependencies(driving_demo rclcpp geometry_msgs nav_msgs)
install(TARGETS driving_demo DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/driving_demo.sh DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/run_demo_in_container.sh DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config worlds DESTINATION share/${PROJECT_NAME})
# 독립 Gazebo 검증용 C++ 실행 파일입니다. 일반 launch에서는 실행하지 않습니다.
if(BUILD_TESTING)
  find_package(ros_gz_interfaces REQUIRED)
  add_executable(vehicle_smoke_test test/vehicle_smoke.cpp)
  ament_target_dependencies(vehicle_smoke_test rclcpp geometry_msgs nav_msgs ros_gz_interfaces)
  install(TARGETS vehicle_smoke_test DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
